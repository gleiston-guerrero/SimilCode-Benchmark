```csharp
using System;

namespace MedidasFormas
{
    class Calculadora
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== Cálculo de Medidas ===");
            double longitud, anchura;
            
            Console.Write("Ingrese la longitud del rectángulo: ");
            longitud = Convert.ToDouble(Console.ReadLine());
            
            Console.Write("Ingrese la anchura del rectángulo: ");
            anchura = Convert.ToDouble(Console.ReadLine());
            
            double superficie = longitud * anchura;
            double contorno = 2 * (longitud + anchura);
            
            Console.WriteLine($"La superficie es: {superficie} unidades cuadradas");
            Console.WriteLine($"El contorno es: {contorno} unidades");
            
            if (superficie > 100)
            {
                Console.WriteLine("Rectángulo amplio");
            }
            else
            {
                Console.WriteLine("Rectángulo estrecho");
            }
            
            Console.ReadKey();
        }
    }
}
```
