using System;
using System.Collections.Generic;

namespace GestorDeStock
{
    class AdministradorInventario
    {
        static void Main(string[] args)
        {
            List<string> items = new List<string>();
            List<int> existencias = new List<int>();
            
            Console.WriteLine("=== SISTEMA DE INVENTARIO ===");
            
            for (int contador = 0; contador < 3; contador++)
            {
                Console.Write($"Producto {contador + 1}: ");
                string articulo = Console.ReadLine();
                items.Add(articulo);
                
                Console.Write($"Cantidad de {articulo}: ");
                int cantidad = Convert.ToInt32(Console.ReadLine());
                existencias.Add(cantidad);
            }
            
            int sumaTotal = 0;
            Console.WriteLine("\n=== INVENTARIO ACTUAL ===");
            for (int indice = 0; indice < items.Count; indice++)
            {
                Console.WriteLine($"{items[indice]}: {existencias[indice]} unidades");
                sumaTotal += existencias[indice];
                
                if (existencias[indice] < 10)
                    Console.WriteLine("  ⚠️ Stock bajo");
                else if (existencias[indice] > 50)
                    Console.WriteLine("  ✅ Stock alto");
                else
                    Console.WriteLine("  ➡️ Stock normal");
            }
            
            Console.WriteLine($"\nTotal de artículos en inventario: {sumaTotal}");
            
            if (sumaTotal < 50)
                Console.WriteLine("Inventario necesita reposición");
            else
                Console.WriteLine("Inventario en buen estado");
        }
    }
}
