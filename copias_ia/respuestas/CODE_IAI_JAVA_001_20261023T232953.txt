package Java.Orginal;

public class NotasCalculadora {
    public static void main(String[] args) {
        int[] resultados = {85, 90, 78, 92, 88};
        int total = 0;
        
        for(int indice = 0; indice < resultados.length; indice++) {
            total += resultados[indice];
        }
        
        double media = total / (double)resultados.length;
        
        if(media >= 90) {
            System.out.println("Excelente: " + media);
        } else if(media >= 80) {
            System.out.println("Bueno: " + media);
        } else {
            System.out.println("Regular: " + media);
        }
    }
}
