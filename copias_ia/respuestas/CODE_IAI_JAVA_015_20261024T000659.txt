package Java.NuevoProyecto;

import java.util.Scanner;

public class AdministradorDeTareas {
    public static void main(String[] args) {
        Scanner lector = new Scanner(System.in);
        String[] listaTareas = new String[10];
        int indice = 0;
        
        while (true) {
            System.out.println("\n--- GESTOR DE TAREAS ---");
            System.out.println("1. Incluir tarea");
            System.out.println("2. Ver tareas");
            System.out.println("3. Terminar");
            System.out.print("Elige una opción: ");
            
            int seleccion = lector.nextInt();
            lector.nextLine(); // Limpiar el buffer
            
            if (seleccion == 1) {
                if (indice < 10) {
                    System.out.print("Introduce la tarea: ");
                    listaTareas[indice] = lector.nextLine();
                    indice++;
                    System.out.println("Tarea añadida!");
                } else {
                    System.out.println("Lista completa!");
                }
            } else if (seleccion == 2) {
                System.out.println("\n--- TAREAS REGISTRADAS ---");
                if (indice == 0) {
                    System.out.println("No hay tareas registradas");
                } else {
                    for (int i = 0; i < indice; i++) {
                        System.out.println((i + 1) + ". " + listaTareas[i]);
                    }
                }
            } else if (seleccion == 3) {
                System.out.println("¡Adiós!");
                break;
            } else {
                System.out.println("Opción no válida");
            }
        }
        lector.close();
    }
}
