using System;
using System.Collections.Generic;

namespace AdminAlumnos
{
    class Control
    {
        static void Main(string[] args)
        {
            List<string> alumnos = new List<string>();
            List<double> notas = new List<double>();
            
            Console.WriteLine("=== SISTEMA DE ALUMNOS ===");
            
            for (int i = 0; i < 3; i++)
            {
                Console.Write($"Ingrese nombre del alumno {i + 1}: ");
                string nombre = Console.ReadLine();
                alumnos.Add(nombre);
                
                Console.Write($"Ingrese calificación de {nombre}: ");
                double calificacion = Convert.ToDouble(Console.ReadLine());
                notas.Add(calificacion);
            }
            
            double total = 0;
            for (int i = 0; i < notas.Count; i++)
            {
                total += notas[i];
                Console.WriteLine($"{alumnos[i]}: {notas[i]}");
            }
            
            double promedio = total / notas.Count;
            Console.WriteLine($"Promedio del grupo: {promedio:F2}");
            
            if (promedio >= 7.0)
                Console.WriteLine("Excelente rendimiento grupal");
            else
                Console.WriteLine("Necesita mejorar el grupo");
        }
    }
}
