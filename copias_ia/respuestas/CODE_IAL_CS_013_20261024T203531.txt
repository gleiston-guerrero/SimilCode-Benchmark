```csharp
using System;

namespace UnidadesTransformer
{
    class Transformador
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== TRANSFORMADOR DE UNIDADES ===");
            Console.WriteLine("1. Metros a Pies");
            Console.WriteLine("2. Celsius a Fahrenheit");
            Console.WriteLine("3. Kilogramos a Libras");
            Console.Write("Seleccione transformación: ");
            
            int seleccion = Convert.ToInt32(Console.ReadLine());
            double valorFinal = 0;
            
            if (seleccion == 1)
            {
                Console.Write("Ingrese metros: ");
                double metros = Convert.ToDouble(Console.ReadLine());
                valorFinal = metros * 3.28084;
                Console.WriteLine($"{metros} metros = {valorFinal:F2} pies");
            }
            else if (seleccion == 2)
            {
                Console.Write("Ingrese grados Celsius: ");
                double celsius = Convert.ToDouble(Console.ReadLine());
                valorFinal = (celsius * 9/5) + 32;
                Console.WriteLine($"{celsius}°C = {valorFinal:F2}°F");
            }
            else if (seleccion == 3)
            {
                Console.Write("Ingrese kilogramos: ");
                double kg = Convert.ToDouble(Console.ReadLine());
                valorFinal = kg * 2.20462;
                Console.WriteLine($"{kg} kg = {valorFinal:F2} libras");
            }
            
            if (valorFinal > 100)
                Console.WriteLine("Valor transformado alto");
            else
                Console.WriteLine("Valor transformado normal");
        }
    }
}
```
