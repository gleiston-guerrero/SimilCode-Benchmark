```csharp
using System;

namespace TransformadorDeMedidas
{
    class Aplicacion
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== TRANSFORMADOR DE MEDIDAS ===");
            Console.WriteLine("1. Metros a Pies");
            Console.WriteLine("2. Celsius a Fahrenheit");
            Console.WriteLine("3. Kilogramos a Libras");
            Console.Write("Seleccione conversión: ");
            
            int seleccion = Convert.ToInt32(Console.ReadLine());
            double conversion = 0;
            
            if (seleccion == 1)
            {
                Console.Write("Ingrese metros: ");
                double metros = Convert.ToDouble(Console.ReadLine());
                conversion = metros * 3.28084;
                Console.WriteLine($"{metros} metros = {conversion:F2} pies");
            }
            else if (seleccion == 2)
            {
                Console.Write("Ingrese grados Celsius: ");
                double celsius = Convert.ToDouble(Console.ReadLine());
                conversion = (celsius * 9/5) + 32;
                Console.WriteLine($"{celsius}°C = {conversion:F2}°F");
            }
            else if (seleccion == 3)
            {
                Console.Write("Ingrese kilogramos: ");
                double kg = Convert.ToDouble(Console.ReadLine());
                conversion = kg * 2.20462;
                Console.WriteLine($"{kg} kg = {conversion:F2} libras");
            }
            
            if (conversion > 100)
                Console.WriteLine("Valor convertido alto");
            else
                Console.WriteLine("Valor convertido normal");
        }
    }
}
```
