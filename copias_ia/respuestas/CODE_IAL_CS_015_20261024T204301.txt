```csharp
using System;
using System.Collections.Generic;

namespace AnalizadorDeDatos
{
    class Ejecutor
    {
        static void Main()
        {
            List<int> valores = new List<int>();
            
            Console.WriteLine("Analizador de Datos");
            Console.Write("¿Cuántos valores desea analizar? ");
            int cantidad = int.Parse(Console.ReadLine());
            
            for (int i = 0; i < cantidad; i++)
            {
                Console.Write($"Ingrese el valor {i + 1}: ");
                int valor = int.Parse(Console.ReadLine());
                valores.Add(valor);
            }
            
            int total = 0;
            int max = valores[0];
            int min = valores[0];
            
            foreach (int val in valores)
            {
                total += val;
                if (val > max) max = val;
                if (val < min) min = val;
            }
            
            double media = (double)total / valores.Count;
            
            Console.WriteLine($"\nResultados:");
            Console.WriteLine($"Suma: {total}");
            Console.WriteLine($"Promedio: {media:F2}");
            Console.WriteLine($"Máximo: {max}");
            Console.WriteLine($"Mínimo: {min}");
            Console.WriteLine($"Rango: {max - min}");
            
            if (media > 50)
                Console.WriteLine("Los valores tienden a ser altos");
            else
                Console.WriteLine("Los valores tienden a ser bajos");
        }
    }
}
```
